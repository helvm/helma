   16 runAsRIO :: (MonadIO m, MonadReader env m, Has env) => LoggingT (SafeT m) a -> m a
   17 runAsRIO action = do
   18   logFunc <- RIO.view RIO.logFuncL
   19   let logOutput _ source level msg =  RIO.runRIO logFunc $ RIO.logGeneric source (toRioLevel level) (RIO.displayBytesUtf8 $ fromLogStr msg)
   20   result <- runExceptT $ runLoggingT action logOutput
   21   either ((*> RIO.exitFailure) . RIO.logError . RIO.display . errorsToText) pure result
