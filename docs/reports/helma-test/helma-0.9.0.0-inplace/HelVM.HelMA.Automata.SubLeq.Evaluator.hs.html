   33 runRio ∷ Has env ⇒ RIO.RIO env ()
   34 runRio = runWithOptions =<< optionsRio where
   35   runWithOptions o =  run (App.emit o) . App.evalParams o =<< readSourceFileRio

   37 run ∷ Has env ⇒ Emit.Emit → EvalParams → RIO.RIO env ()

   48 evalParams ∷ AppSafeEff m ⇒ EvalParams → m ()

   51 evalSource ∷ AppSafeEff m ⇒ Source → RAMType → LimitMaybe → DumpType → m ()
   52 evalSource source = evalIL $ tokenize source

   54 evalIL ∷ AutomatonEff e m ⇒ [e] → RAMType → LimitMaybe → DumpType → m ()
   55 evalIL = flip evalIL'
   56 
   57 evalIL' ∷ AutomatonEff e m ⇒ RAMType → [e] → LimitMaybe → DumpType → m ()
   58 evalIL' ListRAMType    = start
   59 evalIL' SeqRAMType     = start . Seq.fromList
   60 evalIL' SListRAMType   = start . SList.sListFromList
   61 evalIL' MapListRAMType = start . Seq.fromList

   63 start ∷ RAutomatonEff e r m ⇒ r → LimitMaybe → DumpType → m ()
