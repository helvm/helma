   34 runRio :: Has env => RIO.RIO env ()
   35 runRio = runWithOptions =<< optionsRio where
   36   runWithOptions o =  run (App.emit o) . App.evalParams o =<< readSourceFileRio

   38 run :: Has env => Emit.Emit -> EvalParams -> RIO.RIO env ()
   39 run Emit.No   = runAsRIO . evalParams
   40 run Emit.IL   = putLTextLnRio . show . tokenize . source
   41 run Emit.TL   = putLTextLnRio . show . tokenize . source
   42 run Emit.Code = putLTextLnRio . show . readSymbols . source

   52 evalSource :: AppEff m => Source -> RAMType -> LimitMaybe -> DumpType -> m ()
   53 evalSource source = evalIL $ tokenize source
   54 
   55 evalIL :: AutomatonEff e m => [e] -> RAMType -> LimitMaybe -> DumpType -> m ()
   56 evalIL = flip evalIL'
   57 
   58 evalIL' :: AutomatonEff e m => RAMType -> [e] -> LimitMaybe -> DumpType -> m ()
   59 evalIL' ListRAMType    = start
   60 evalIL' SeqRAMType     = start . Seq.fromList
   61 evalIL' SListRAMType   = start . SList.sListFromList
   62 evalIL' MapListRAMType = start . MapList.mapListFromList
   63 
   64 start :: RAutomatonEff e r m => r -> LimitMaybe -> DumpType -> m ()
   65 start r limit dt = logDump dt =<< runAutomat limit (newMemory r)
