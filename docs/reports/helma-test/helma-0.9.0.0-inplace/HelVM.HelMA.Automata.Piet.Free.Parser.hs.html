   20 parseColorMap ∷ CodelSize → DynamicImage → Either ProgramError ColorMap
   21 parseColorMap cs dyn = imageToColorMap cs <$> toRGB8 dyn
   22 
   23 imageToColorMap ∷ CodelSize → Image PixelRGB8 → ColorMap
   24 imageToColorMap cs img = ColorMap matrix' w' (V.length matrix')
   25   where matrix' = to2D w' . V.fromList $ map (pixelToColor . pixAt) coords
   26         coords = [(x, y) | y <- [0..Extra.prev h], x <- [0..Extra.prev w], cs |^ x, cs |^ y]
   27         w = imageWidth img
   28         h = imageHeight img
   29         w' = w `div` cs
   30         a |^ b = b `mod` a == 0
   31         pixAt = uncurry (pixelAt img)
   32 
   33 toRGB8 ∷ DynamicImage → Either ProgramError (Image PixelRGB8)
   34 toRGB8 = first NotImplemented . toRGB8OrError
   35 
   36 to2D ∷ Int → V.Vector a → V.Vector (V.Vector a)
   37 to2D width v
   38   | V.null v = V.empty
   39   | otherwise = uncurry collect $ V.splitAt width v
   40   where
   41     collect h t
   42       | V.length t < width = V.singleton h
   43       | otherwise          = V.cons h (to2D width t)
