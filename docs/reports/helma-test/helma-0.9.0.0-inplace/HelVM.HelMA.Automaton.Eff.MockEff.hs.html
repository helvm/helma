   68 runMockEff :: Input -> MockEff (Safe ()) -> MockEffData
   69 runMockEff i mockIO = safeToMockData $ runWriter $ runStateT (unMockEff mockIO) $ createMockIOData i where
   70   safeToMockData ((Right _, io), logs) = MockEffData io logs
   71   safeToMockData ((Left msgs, io), logs) = MockEffData io (logs Seq.|> errLog) where
   72     errLog = MockLog defaultLoc "" LevelError $ toLogStr $ errorsToText msgs

  115 mockGetContents :: MonadMockEff m => m String
  116 mockGetContents = mockGetContents' =<< get where
  117   mockGetContents' :: MonadMockEff m => MockIOData -> m String
  118   mockGetContents' mockIO = content <$ put mockIO { input = "" } where content = input mockIO

  120 mockGetChar :: MonadMockEff m => m Char
  121 mockGetChar = mockGetChar' =<< get where
  122   mockGetChar' :: MonadMockEff m => MockIOData -> m Char
  123   mockGetChar' mockIO = orErrorTuple ("mockGetChar" , Text.show mockIO) (top (input mockIO)) <$ put mockIO { input = orErrorTuple ("mockGetChar" , Text.show mockIO) $ discard $ input mockIO }
  124 
  125 mockGetLine :: MonadMockEff m => m Text
  126 mockGetLine = mockGetLine' =<< get where
  127   mockGetLine' :: MonadMockEff m => MockIOData -> m Text
  128   mockGetLine' mockIO = toText line <$ put mockIO { input = input' } where (line , input') = splitStringByLn $ input mockIO

  130 mockGetCharSafe :: MonadSafeMockEff m => m Char

  134     mockGetChar'' (c, input') = put mockIO { input = input' } $> c
