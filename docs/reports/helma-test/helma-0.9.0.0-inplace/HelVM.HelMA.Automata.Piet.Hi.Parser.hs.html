   18 parseColorMap cs dyn = imageToColorMap cs <$> toRGB8 dyn

   20 imageToColorMap ∷ CodelSize → Image PixelRGB8 → ColorMap
   21 imageToColorMap cs img = ColorMap matrix' w' (V.length matrix')
   22   where matrix' = to2D w' . V.fromList $ map (pixelToColor . pixAt) coords
   23         coords = [(x, y) | y <- [0..Extra.prev h], x <- [0..Extra.prev w], cs |^ x, cs |^ y]
   24         w = imageWidth img
   25         h = imageHeight img
   26         w' = w `div` cs
   27         a |^ b = b `mod` a == 0
   28         pixAt = uncurry (pixelAt img)

   49 to2D ∷ Int → V.Vector a → V.Vector (V.Vector a)
   50 to2D width v
   51   | V.null v = V.empty
   52   | otherwise = uncurry collect $ V.splitAt width v
   53   where
   54     collect h t
   55       | V.length t < width = V.singleton h
   56       | otherwise          = V.cons h (to2D width t)
