   57 runMockEff ∷ Input → Mock (Safe ()) → MockData
   58 runMockEff i mockEff = safeToMockData $ runWriter $ runStateT (unMock mockEff) $ createMockEffData i where
   59   safeToMockData ((Right _, io), logs) = (io, logs)
   60   safeToMockData ((Left msgs, io), logs) = (io, logs Seq.|> errLog) where
   61     errLog = MockLog defaultLoc "" LevelError $ toLogStr $ errorsToText msgs
