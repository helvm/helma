   37 runRio ∷ Has env ⇒ TokenType → RIO.RIO env ()
   38 runRio t = runWithOptions =<< optionsRio where
   39   runWithOptions o = run (App.emit o) t . App.evalParams o =<< readSourceFileRio
   40 
   41 run ∷ Has env ⇒ Emit → TokenType → EvalParams → RIO.RIO env ()
   42 run No   t                = runAsRIO . evalParams t
   43 run IL   VisibleTokenType = putLTextLnRio . pShowNoColor . (flipParseVisible <$> formatType <*> source)
   44 run IL   WhiteTokenType   = putLTextLnRio . pShowNoColor . (flipParseWhite   <$> formatType <*> source)
   45 run TL   VisibleTokenType = putLTextLnRio . show . tokenizeVisible . source
   46 run TL   WhiteTokenType   = putLTextLnRio . show . tokenizeWhite   . source
   47 run Code VisibleTokenType = putLTextLnRio . show . readVisibleTokens . source
   48 run Code WhiteTokenType   = putLTextLnRio . show . readWhiteTokens   . source

   51 simpleEval ∷ AppEff m ⇒ S.SimpleParams → m ()
   52 simpleEval p = eval (S.tokenType p) (S.source p) (S.formatType p) $ S.automatonOptions p

   56 evalParams ∷ AppEff m ⇒ TokenType → EvalParams → m ()
   57 evalParams tokenType p = eval tokenType (source p) (formatType p) $ automatonOptions p

   59 eval ∷ AppEff m ⇒ TokenType → Source → LabelType → Automaton.AutomatonOptions → m ()
   60 eval tokenType source = evalTL $ tokenize tokenType source

   62 evalTL ∷ AppEff m ⇒ TokenList → LabelType → Automaton.AutomatonOptions → m ()
   63 evalTL tl ascii ao = flip start ao =<< liftSafe (parseFromTL ascii tl)
