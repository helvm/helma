   87 type PietT m a = ExceptT ProgramError (ReaderT ProgramConfig (StateT ProgramState m)) a
   88 type Piet a = PietT IO a

   90 initialState ∷ ProgramState
   91 initialState = ProgramState
   92   { _directionPointer = DPRight
   93   , _codelChooser     = CCLeft
   94   , _currentPosition  = (0, 0)
   95   , _stack            = []
   96   , _collisionCount   = 0
   97   }

   99 runPiet ∷ ProgramConfig → ProgramState → PietT m a → m (Either ProgramError a, ProgramState)
  100 runPiet conf s action = runStateT (runReaderT (runExceptT action) conf) s

  104 instance RIO.Display ProgramError where

  106   textDisplay (LoadFile m)       = "Error while loading file: " <> m
  107   textDisplay (FindFile m)       = "Can't find file: " <> m
