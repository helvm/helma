   39 doBinary ∷ Integral a ⇒ BinaryOperation → a → a → a
   40 doBinary Add = (+)
   41 doBinary Sub = (-)
   42 doBinary Mul = (*)
   43 doBinary Div = div
   44 doBinary Mod = mod
   45 doBinary LGT = lGT

  101 printSPure ∷ SPureInstruction → Text
  102 printSPure (Unary    i  ) = printUnary i
  103 printSPure (Indexed  i o) = toLowerShow o <> printIndexOperand i
  104 printSPure (Binary   i  ) = toLowerShow i
  105 printSPure (Binaries i  ) = printBinaries i
  106 printSPure           i    = toLowerShow i

  108 printBinaries ∷ (Foldable c, Functor c, Show i) ⇒ c i → Text
  109 printBinaries il = fmconcat $ toLowerShow <$> il

  111 printUnary ∷ UnaryOperation → Text
