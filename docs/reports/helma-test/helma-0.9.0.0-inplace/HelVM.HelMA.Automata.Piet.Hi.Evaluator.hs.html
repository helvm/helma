   32 execute ∷ MonadLoggerEff m ⇒ CodelSize → ColorMap → m ()
   33 execute cs cmap = void $ runPiet conf initialState loop
   34   where
   35     conf = ProgramConfig { codelSize = cs, colorMap = cmap }
   36     loop = transition >>= (`when` loop)

   38 interpret ∷ MonadLoggerEff m ⇒ Program → PietT m ()
   39 interpret (Pure _)                  = pass
   40 interpret (Free (InstructionF i r)) = evalInstruction i r
   41 
   42 evalInstruction ∷ MonadLoggerEff m ⇒ Instruction → Program → PietT m ()
   43 evalInstruction (Push n)  r = logMsg ("push " <> show n) >> stack %= (n:) >> interpret r
   44 evalInstruction Pop       r = logMsg "pop" >> stack %= popStack >> interpret r
   45 evalInstruction Add       r = logMsg "add" >> perform r (+)
   46 evalInstruction Subtract  r = logMsg "subtract" >> perform r (-)
   47 evalInstruction Multiply  r = logMsg "multiply" >> perform r (*)
   48 evalInstruction Divide    r = logMsg "divide" >> perform r div
   49 evalInstruction Mod       r = logMsg "mod" >> perform r mod
   50 evalInstruction Not       r = logMsg "not" >> stack %= stepNot >> interpret r
   51 evalInstruction Greater   r = logMsg "greater" >> perform r greater'
   52 evalInstruction Pointer   r = logMsg "pointer" >> flip' 4 rotatePointer >> interpret r
   53 evalInstruction Switch    r = logMsg "switch" >> flip' 2 toggleChooser >> interpret r
   54 evalInstruction Duplicate r = logMsg "duplicate" >> stack %= dupStack >> interpret r
   55 evalInstruction Roll      r = logMsg "roll" >> performRoll >> interpret r
   56 evalInstruction InNum     r = logMsg "innum" >> whenJustM readInt' pushInt >> interpret r
   57 evalInstruction InChar    r = logMsg "inchar" >> (getCharAsInt >>= pushChar) >> interpret r
   58 evalInstruction OutNum    r = logMsg "out_num" >> printTop putIntAsDec >> interpret r
   59 evalInstruction OutChar   r = logMsg "out_char" >> printTop putIntAsCharSafe >> interpret r
   60 evalInstruction Nop       r = interpret r

   62 logMsg ∷ MonadLoggerEff m ⇒ Text → PietT m ()
   63 logMsg msg = do
   64   (x, y) <- use currentPosition
   65   logDebugN $ "(" <> show x <> "," <> show y <> ") " <> msg

   67 popStack ∷ [a] → [a]
   68 popStack [] = []
   69 popStack s  = tailUnsafe s
   70 
   71 dupStack ∷ [a] → [a]
   72 dupStack [] = []
   73 dupStack s  = headUnsafe s : s
   74 
   75 manip ∷ (a → a → a) → [a] → [a]
   76 manip op (a : b : cs) = (b `op` a) : cs
   77 manip _  s            = s
   78 
   79 perform ∷ MonadLoggerEff m ⇒ Program → (Int → Int → Int) → PietT m ()
   80 perform r op = stack %= manip op >> interpret r
   81 
   82 greater' ∷ Int → Int → Int
   83 b `greater'` a
   84   | b > a     = 1
   85   | otherwise = 0
   86 
   87 stepNot ∷ [Int] → [Int]
   88 stepNot (x : xs) = not'' x : xs
   89 stepNot []       = []
   90 
   91 not'' ∷ Int → Int
   92 not'' 0 = 1
   93 not'' _ = 0

   95 flip' ∷ Monad m ⇒ Int → PietT m () → PietT m ()
   96 flip' n f = do
   97   s <- use stack
   98   unless (null s) $
   99     replicateM_ (headUnsafe s `mod` n) f

  101 readInt' ∷ MonadEff m ⇒ PietT m (Maybe Int)
  102 readInt' = readMaybe . toString <$> getLine

  104 pushInt ∷ Monad m ⇒ Int → PietT m ()
  105 pushInt = (stack %=) . (:)
  106 
  107 pushChar ∷ Monad m ⇒ Int → PietT m ()
  108 pushChar = pushInt

  110 printTop ∷ Monad m ⇒ (Int → PietT m a) → PietT m ()
  111 printTop f = do
  112   s <- use stack
  113   unless (null s) $ do
  114     _ <- f (headUnsafe s)
  115     stack %= tailUnsafe

  117 putIntAsCharSafe ∷ MonadEff m ⇒ Int → PietT m ()
  118 putIntAsCharSafe n = putIntAsChar (n `mod` 256)
  119 
  120 performRoll ∷ Monad m ⇒ PietT m ()
  121 performRoll = do
  122   s <- use stack
  123   rollStack s

  125 rollStack ∷ Monad m ⇒ [Int] → PietT m ()
  126 rollStack (times : depth : rest)
  127   | depth >= 0 = stack .= roll' depth times rest
  128 rollStack _ = pass
  129 
  130 roll' ∷ Int → Int → [Int] → [Int]
  131 roll' depth times rest
  132   | depth <= 0 = rest
  133   | otherwise  = checkLength (splitAt depth rest)
  134   where
  135     checkLength (h, t)
  136       | olength h < depth = rest
  137       | otherwise        = rotateCycle (times `mod` olength h) h t
  138 
  139 rotateCycle ∷ Int → [a] → [a] → [a]
  140 rotateCycle n h t
  141   | n < 0     = take (olength h) (drop (n + olength h) (cycle h)) <> t
  142   | otherwise = take (olength h) (drop n (cycle h)) <> t

  144 onMap ∷ ColorMap → Coordinates → Bool
  145 onMap m (i, j) =
  146   let w = _mapWidth m
  147       h = _mapHeight m
  148   in (0 <= i && i < w) && (0 <= j && j < h)
  149 
  150 infixl 9 &!
  151 (&!) ∷ ColorMap → Coordinates → Maybe Color
  152 m &! c = getPixel m (onMap m c) c
  153 
  154 getPixel ∷ ColorMap → Bool → Coordinates → Maybe Color
  155 getPixel m True  (i, j) = Just $ _matrix m ! j ! i
  156 getPixel _ False _      = Nothing

  158 discoverBlock ∷ ColorMap → Coordinates → Block
  159 discoverBlock m startPos = S.toList $ go S.empty startPos
  160   where
  161     targetColor = m &! startPos
  162 
  163     go visited pos
  164       | pos `S.member` visited = visited
  165       | m &! pos /= targetColor = visited
  166       | otherwise = L.foldl' go (S.insert pos visited) (neighbours pos)
  167 
  168 neighbours ∷ Coordinates → [Coordinates]
  169 neighbours (x, y) = [(x, Extra.prev y), (x, Extra.next y), (Extra.prev x, y), (Extra.next x, y)]

  171 lightnessSteps ∷ Color → Color → Maybe Int
  172 lightnessSteps (Chromatic (ChromaticColor l1 _)) (Chromatic (ChromaticColor l2 _)) =
  173   Just $ (fromEnum l2 - fromEnum l1) `mod` 3
  174 lightnessSteps _ _ = Nothing
  175 
  176 hueSteps ∷ Color → Color → Maybe Int
  177 hueSteps (Chromatic (ChromaticColor _ h1)) (Chromatic (ChromaticColor _ h2)) =
  178   Just $ (fromEnum h2 - fromEnum h1) `mod` 6
  179 hueSteps _ _ = Nothing

  181 colorsToInstruction ∷ Color → Color → Int → Instruction
  182 colorsToInstruction c c' = step (lightnessSteps c c') (hueSteps c c')
  183 
  184 step ∷ Maybe Int → Maybe Int → Int → Instruction
  185 step (Just 0) (Just 0) _ = Nop
  186 step (Just 0) (Just 1) _ = Add
  187 step (Just 0) (Just 2) _ = Divide
  188 step (Just 0) (Just 3) _ = Greater
  189 step (Just 0) (Just 4) _ = Duplicate
  190 step (Just 0) (Just 5) _ = InChar
  191 step (Just 1) (Just 0) n = Push n
  192 step (Just 1) (Just 1) _ = Subtract
  193 step (Just 1) (Just 2) _ = Mod
  194 step (Just 1) (Just 3) _ = Pointer
  195 step (Just 1) (Just 4) _ = Roll
  196 step (Just 1) (Just 5) _ = OutNum
  197 step (Just 2) (Just 0) _ = Pop
  198 step (Just 2) (Just 1) _ = Multiply
  199 step (Just 2) (Just 2) _ = Not
  200 step (Just 2) (Just 3) _ = Switch
  201 step (Just 2) (Just 4) _ = InNum
  202 step (Just 2) (Just 5) _ = OutChar
  203 step _        _        _ = Nop

  205 colorsToProgram ∷ Color → Color → Int → Program
  206 colorsToProgram c c' n = liftF $ InstructionF (colorsToInstruction c c' n) ()

  208 transition ∷ MonadLoggerEff m ⇒ PietT m Bool
  209 transition = do
  210   cc <- use collisionCount
  211   transitionStep cc
  212 
  213 transitionStep ∷ MonadLoggerEff m ⇒ Int → PietT m Bool
  214 transitionStep cc
  215   | cc >= 8   = logDebugN "Max collisions reached (8). Terminating." >> pure False
  216   | otherwise = do
  217       dp <- use directionPointer
  218       pos <- use currentPosition
  219       m <- asks colorMap
  220       let block = discoverBlock m pos
  221       p <- selectCodel block
  222       let newPos = move dp p
  223       colour <- colourAt newPos
  224       handleNextColour pos newPos block colour
  225 
  226 handleNextColour ∷ MonadLoggerEff m ⇒ Coordinates → Coordinates → Block → Maybe Color → PietT m Bool
  227 handleNextColour _ _ _ Nothing           = doIfCollided >> pure True
  228 handleNextColour _ _ _ (Just Black)     = doIfCollided >> pure True
  229 handleNextColour _ newPos _ (Just White) = do
  230   collisionCount .= 0
  231   currentPosition .= newPos
  232   pure True
  233 handleNextColour pos newPos block (Just c') = do
  234   mbC <- colourAt pos
  235   collisionCount .= 0
  236   currentPosition .= newPos
  237   evalTransitionBlock pos c' block mbC
  238   pure True
  239 
  240 evalTransitionBlock ∷ MonadLoggerEff m ⇒ Coordinates → Color → Block → Maybe Color → PietT m ()
  241 evalTransitionBlock _ c' block (Just c) | c /= White = do
  242   cs <- asks codelSize
  243   let numCodels = olength block `div` (cs * cs)
  244       instr = colorsToProgram c c' numCodels
  245   interpret instr
  246 evalTransitionBlock _ _ _ _ = pass
  247 
  248 doIfCollided ∷ Monad m ⇒ PietT m ()
  249 doIfCollided = do
  250   cc <- use collisionCount
  251   handleCollision (even cc)
  252   collisionCount += 1
  253 
  254 handleCollision ∷ Monad m ⇒ Bool → PietT m ()
  255 handleCollision True  = toggleChooser
  256 handleCollision False = rotatePointer
  257 
  258 colourAt ∷ Monad m ⇒ Coordinates → PietT m (Maybe Color)
  259 colourAt pos = (&! pos) <$> asks colorMap
  260 
  261 toggleChooser ∷ Monad m ⇒ PietT m ()
  262 toggleChooser = codelChooser %= nextChooser
  263 
  264 rotatePointer ∷ Monad m ⇒ PietT m ()
  265 rotatePointer = do
  266   dp <- use directionPointer
  267   directionPointer .= nextPointer dp
  268 
  269 selectCodel ∷ Monad m ⇒ Block → PietT m Coordinates
  270 selectCodel block = do
  271   dp <- use directionPointer
  272   cc <- use codelChooser
  273   return $ L.maximumBy (Orientation.furthest cc dp) block

  275 headUnsafe ∷ [a] → a
  276 headUnsafe []      = error "empty list"
  277 headUnsafe (h : _) = h
  278 
  279 tailUnsafe ∷ [a] → [a]
  280 tailUnsafe []      = error "empty list"
  281 tailUnsafe (_ : t) = t
