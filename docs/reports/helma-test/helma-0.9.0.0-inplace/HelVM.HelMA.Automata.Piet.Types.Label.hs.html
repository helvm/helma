   51 addPixel ∷ Coordinates → Maybe LabelInfo → Maybe LabelInfo
   52 addPixel (x, y) Nothing = Just $ LabelInfo 1 (LabelBorder y x x) (LabelBorder x y y) (LabelBorder y x x) (LabelBorder x y y)
   53 addPixel (x, y) (Just stats) = Just $ stats
   54   & labelSize %~ (+ 1)
   55   & labelTop %~ (`mergeMin` LabelBorder y x x)
   56   & labelLeft %~ (`mergeMin` LabelBorder x y y)
   57   & labelBottom %~ (`mergeMax` LabelBorder y x x)
   58   & labelRight %~ (`mergeMax` LabelBorder x y y)

   60 getLabelSize ∷ Maybe LabelInfo → Int
   61 getLabelSize Nothing     = 0

   64 instance Semigroup LabelInfo where

   68     (mergeMin (s1 ^. labelLeft) (s2 ^. labelLeft))

   74 mergeMin ∷ LabelBorder → LabelBorder → LabelBorder
   75 mergeMin = merge $ comparing (^. borderCoord)
   76 
   77 mergeMax ∷ LabelBorder → LabelBorder → LabelBorder
   78 mergeMax = merge $ comparing (negate . (^. borderCoord))

   80 merge ∷ (LabelBorder → LabelBorder → Ordering) → LabelBorder → LabelBorder → LabelBorder
   81 merge comp b1 b2 = go $ comp b1 b2 where
   82   go EQ = b1
   83     & borderMin %~ min (b2 ^. borderMin)
   84     & borderMax %~ max (b2 ^. borderMax)
   85   go LT = b1
   86   go GT = b2
