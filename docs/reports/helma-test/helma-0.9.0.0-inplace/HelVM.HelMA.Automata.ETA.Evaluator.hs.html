   47 runRio ∷ Has env ⇒ AutomatonType → RIO.RIO env ()
   48 runRio i = runWIthOptions =<< optionsRio where
   49   runWIthOptions o = run (App.emit o) i . App.evalParams o =<< readSourceFileRio
   50 
   51 run ∷ Has env ⇒ Emit.Emit → AutomatonType → EvalParams → RIO.RIO env ()
   52 run Emit.No   i = runAsRIO . evalParams i
   53 run Emit.IL   _ = putLTextLnRio . pShowNoColor . parseSafe . source
   54 run Emit.TL   _ = putLTextLnRio . show . tokenize . source
   55 run Emit.Code _ = putLTextLnRio . show . readTokens . source

   57 simpleEval ∷ AppEff m ⇒ S.SimpleParams → m ()
   58 simpleEval p = evalSource (S.implType p) (S.source p) (S.stackType p) (S.autoOptions p)

   62 evalParams ∷ AppEff m ⇒ AutomatonType → EvalParams → m ()
   63 evalParams e p = evalSource e (source p) (stackAutoOptions p) (autoOptions p)
   64 
   65 evalSource ∷ (AutomatonEff Symbol m) ⇒ AutomatonType → Source → StackType → AutoOptions → m ()
   66 evalSource automatonType source = evalTL automatonType (tokenize source)
   67 
   68 evalTL ∷ (AutomatonEff Symbol m) ⇒ AutomatonType → TokenList → StackType → AutoOptions → m ()
   69 evalTL Fast     = fastEval
   70 evalTL Original = originalEval
   71 
   72 fastEval ∷ (AutomatonEff Symbol m) ⇒ TokenList → StackType → AutoOptions → m ()
   73 fastEval tl s a = flip Automaton.start (Automaton.withDefaultRam s a) =<< optimize tl
   74 
   75 originalEval ∷ (AutomatonEff Symbol m) ⇒ TokenList → StackType → AutoOptions → m ()
   76 originalEval tl ListStackType  = eval tl []
   77 originalEval tl SeqStackType   = eval tl Seq.empty
   78 originalEval tl SListStackType = eval tl SList.sListEmpty
   79 
   80 eval ∷ (SAutomatonEff Symbol s m) ⇒ TokenList → s → AutoOptions → m ()
   81 eval tl s (AutoOptions _ limit dt) = logDump dt =<< runAutomat limit (newMemory tl s)
