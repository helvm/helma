    9 isNotJump ∷ Integral e ⇒ BranchTest → e → Bool
   10 isNotJump t = not . isJump t
   11 
   12 isJump ∷ Integral e ⇒ BranchTest → e → Bool
   13 isJump NE  e = e /= 0
   14 isJump EZ  e = e == 0
   15 isJump LTZ e = e <  0
   16 isJump GTZ e = e >  0

   18 -- | Types
   19 data CFInstruction
   20   = Mark !Mark

   23   | Return
   24   deriving stock (Eq, Read, Show)

   58 printCF ∷ CFInstruction → Text
   59 printCF (Mark     i  ) = "\nmark" <> printMark i
   60 printCF (Branch i t)   = printBranchTest t <> printBranchOperand i
   61 printCF (Labeled  i o) = toLowerShow o <> printLabelOperand i
   62 printCF           i    = toLowerShow i
   63 
   64 printMark ∷ Mark → Text
   65 printMark (MNatural    i) = "M " <> show i
   66 printMark (MArtificial i) = "A " <> show i
   67 
   68 printBranchTest ∷ BranchTest → Text
   69 printBranchTest t = "b" <> show t
   70 
   71 printBranchOperand ∷ BranchOperand → Text
   72 printBranchOperand  BTop           = ""
   73 printBranchOperand  BSwapped       = "S"
   74 printBranchOperand (BImmediate  i) = "I " <> show i
   75 printBranchOperand (BArtificial i) = "A " <> show i
   76 
   77 printLabelOperand ∷ LabelOperand → Text
   78 printLabelOperand  LTop           = ""
   79 printLabelOperand (LImmediate  i) = "I " <> show i
   80 printLabelOperand (LArtificial i) = "A " <> show i
