   10 type Has env = (HasIO env, HasAppOptions env, RIO.HasLogFunc env)
   11 type HasIO env = (HasFileIO env, HasStdIO env)

   13 readSourceFileRio :: Has env => RIO.RIO env Source
   14 readSourceFileRio = readSourceFileWithOptions =<< optionsRio where
   15   readSourceFileWithOptions = readSourceFile <$> exec <*> file
   16   readSourceFile True = pure . toText
   17   readSourceFile _    = readTextFileRio
   18 
   19 readTextFileRio :: Has env => FilePath -> RIO.RIO env Source
   20 readTextFileRio = (RIO.view fileIOL >>=) . flip readTextFile
   21 
   22 putLTextLnRio :: Has env => LText -> RIO.RIO env ()
   23 putLTextLnRio = (RIO.view stdIOL >>=) . flip stdPutLTextLn
   24 
   25 getContentsTextRio :: Has env => RIO.RIO env LText
   26 getContentsTextRio = RIO.view stdIOL >>= stdGetContentsText
   27 
   28 putLBSLnRio :: Has env => LByteString -> RIO.RIO env ()
   29 putLBSLnRio = (RIO.view stdIOL >>=) . flip stdPutLBSLn
   30 
   31 getContentsBSRio :: Has env => RIO.RIO env LByteString
   32 getContentsBSRio = RIO.view stdIOL >>= stdGetContentsBS
   33 
   34 optionsRio :: Has env => RIO.RIO env AppOptions
   35 optionsRio = RIO.view appOptionsL
   36 
   37 logFuncRio :: Has env => RIO.RIO env RIO.LogFunc
   38 logFuncRio = RIO.view RIO.logFuncL

   40 data Env = Env
   41   { envFileIO  :: FileIO
   42   , envStdIO   :: StdIO
   43   , envOptions :: AppOptions
   44   , envLogFunc :: RIO.LogFunc
   45   }
   46 
   47 data FileIO = FileIO
   48   { readTextFile :: forall env. FilePath -> RIO.RIO env Source
   49   , readImage    :: forall env. FilePath -> RIO.RIO env Picture.DynamicImage
   50   }
   51 
   52 class HasFileIO env where
   53   fileIOL :: RIO.Lens' env FileIO
   54 
   55 instance HasFileIO Env where
   56   fileIOL = RIO.lens envFileIO (\x y -> x { envFileIO = y })
   57 
   58 data StdIO = StdIO
   59   { stdPutLTextLn      :: forall env. LText -> RIO.RIO env ()
   60   , stdGetContentsText :: forall env. RIO.RIO env LText
   61   , stdPutLBSLn        :: forall env. LByteString -> RIO.RIO env ()
   62   , stdGetContentsBS   :: forall env. RIO.RIO env LByteString
   63   }
   64 
   65 class HasStdIO env where

   67 
   68 instance HasStdIO Env where
   69   stdIOL = RIO.lens envStdIO (\x y -> x { envStdIO = y })
   70 
   71 class HasAppOptions env where
