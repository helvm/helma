   27 interpretFreeEffF :: MonadEff m => FreeEffF a -> m a
   28 interpretFreeEffF (GetContentsBS    cd) = cd <$> getContentsBS
   29 interpretFreeEffF (GetContentsText  cd) = cd <$> getContentsText
   30 interpretFreeEffF (GetChar          cd) = cd <$> getChar
   31 interpretFreeEffF (GetLine          cd) = cd <$> getLine
   32 interpretFreeEffF (PutChar        c v ) = putChar      c $> v
   33 interpretFreeEffF (PutText        s v ) = putTextEff   s $> v
   34 interpretFreeEffF (Flush            v ) = flush          $> v
   35 
   36 -- | Instances
   37 instance MonadEff FreeEff where
   38   getContentsBS   = freeGetContentsBS

   40   getChar         = freeGetChar
   41   getLine         = freeGetLine
   42   putChar         = freePutChar
   43   putTextEff      = freePutTextEff

   46 -- | Low level functions
