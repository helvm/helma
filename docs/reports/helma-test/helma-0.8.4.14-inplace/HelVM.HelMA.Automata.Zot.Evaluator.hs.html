   47 evalWithFormat :: MonadSafe m => LabelType -> Source -> LText -> m Output
   48 evalWithFormat BinaryLabel source input = pure $ showFoldable $ evalInternal source input

   51 evalInternal :: Source -> LText -> ExpressionDList
